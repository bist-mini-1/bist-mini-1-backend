# 데이터베이스 스키마 명세 (Database Schema)

## 1. 회원 테이블 (members)

```sql
CREATE TABLE members (
    member_id       NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    login_id        VARCHAR2(50) NOT NULL,
    password        VARCHAR2(255) NOT NULL,
    email           VARCHAR2(100) NOT NULL,
    nickname        VARCHAR2(50) NOT NULL,
    bio             VARCHAR2(1000),
    profile_image   VARCHAR2(500),
    status          VARCHAR2(20) DEFAULT 'ACTIVE' NOT NULL,
    created_at      DATE DEFAULT SYSDATE NOT NULL,
    updated_at      DATE,
    deleted_at      DATE,

    CONSTRAINT uk_members_login_id UNIQUE (login_id),
    CONSTRAINT uk_members_email UNIQUE (email),
    CONSTRAINT uk_members_nickname UNIQUE (nickname),
    CONSTRAINT ck_members_status CHECK (status IN ('ACTIVE', 'DELETED'))
);
```

- `bio`: 자기소개
- `status`: 회원 상태 (ACTIVE, DELETED)

---

## 2. 게시글 테이블 (posts)

```sql
CREATE TABLE posts (
    post_id         NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    member_id       NUMBER NOT NULL,
    title           VARCHAR2(200) NOT NULL,
    content         CLOB NOT NULL,
    view_count      NUMBER DEFAULT 0 NOT NULL,
    like_count      NUMBER DEFAULT 0 NOT NULL,
    comment_count   NUMBER DEFAULT 0 NOT NULL,
    is_public       CHAR(1) DEFAULT 'Y' NOT NULL,
    is_temp         CHAR(1) DEFAULT 'N' NOT NULL,
    is_deleted      CHAR(1) DEFAULT 'N' NOT NULL,
    created_at      DATE DEFAULT SYSDATE NOT NULL,
    updated_at      DATE,
    deleted_at      DATE,

    CONSTRAINT fk_posts_member
        FOREIGN KEY (member_id) REFERENCES members(member_id),

    CONSTRAINT ck_posts_public CHECK (is_public IN ('Y', 'N')),
    CONSTRAINT ck_posts_temp CHECK (is_temp IN ('Y', 'N')),
    CONSTRAINT ck_posts_deleted CHECK (is_deleted IN ('Y', 'N'))
);
```

---

## 3. 댓글 테이블 (comments)

```sql
CREATE TABLE comments (
    comment_id      NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    post_id         NUMBER NOT NULL,
    member_id       NUMBER NOT NULL,
    parent_id       NUMBER,
    content         VARCHAR2(2000) NOT NULL,
    is_deleted      CHAR(1) DEFAULT 'N' NOT NULL,
    created_at      DATE DEFAULT SYSDATE NOT NULL,
    updated_at      DATE,
    deleted_at      DATE,

    CONSTRAINT fk_comments_post
        FOREIGN KEY (post_id) REFERENCES posts(post_id),

    CONSTRAINT fk_comments_member
        FOREIGN KEY (member_id) REFERENCES members(member_id),

    CONSTRAINT fk_comments_parent
        FOREIGN KEY (parent_id) REFERENCES comments(comment_id),

    CONSTRAINT ck_comments_deleted CHECK (is_deleted IN ('Y', 'N'))
);
```

- `parent_id`: 대댓글용 (댓글 부모 ID)

## 4. 태그 테이블 (tags)

```sql
CREATE TABLE tags (
    tag_id          NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    name            VARCHAR2(50) NOT NULL,
    created_at      DATE DEFAULT SYSDATE NOT NULL,

    CONSTRAINT uk_tags_name UNIQUE (name)
);
```

---

## 5. 게시글-태그 연결 테이블 (post_tags)

```sql
CREATE TABLE post_tags (
    post_tag_id     NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    post_id         NUMBER NOT NULL,
    tag_id          NUMBER NOT NULL,

    CONSTRAINT fk_post_tags_post
        FOREIGN KEY (post_id) REFERENCES posts(post_id),

    CONSTRAINT fk_post_tags_tag
        FOREIGN KEY (tag_id) REFERENCES tags(tag_id),

    CONSTRAINT uk_post_tags UNIQUE (post_id, tag_id)
);
```

## 8. 첨부파일 테이블 (attachments)

```sql
CREATE TABLE attachments (
    attachment_id   NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    post_id         NUMBER,
    original_name   VARCHAR2(255) NOT NULL,
    file_size       NUMBER NOT NULL,
    file_type       VARCHAR2(100),
    extension       VARCHAR2(20),
    upload_type     VARCHAR2(20) NOT NULL,
    file_data       BLOB NOT NULL,
    download_count  NUMBER DEFAULT 0 NOT NULL,
    is_deleted      CHAR(1) DEFAULT 'N' NOT NULL,
    created_at      DATE DEFAULT SYSDATE NOT NULL,
    deleted_at      DATE,

    CONSTRAINT fk_attachments_post
        FOREIGN KEY (post_id) REFERENCES posts(post_id),

    CONSTRAINT ck_attachments_upload_type
        CHECK (upload_type IN ('THUMBNAIL', 'IMAGE', 'FILE')),

    CONSTRAINT ck_attachments_deleted
        CHECK (is_deleted IN ('Y', 'N'))
);
```

- `extension`: 확장자

## 9. 팔로우 테이블 (follows)

```sql
CREATE TABLE follows (
    follow_id       NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    follower_id     NUMBER NOT NULL,
    following_id    NUMBER NOT NULL,
    created_at      DATE DEFAULT SYSDATE NOT NULL,

    CONSTRAINT fk_follows_follower
        FOREIGN KEY (follower_id) REFERENCES members(member_id),

    CONSTRAINT fk_follows_following
        FOREIGN KEY (following_id) REFERENCES members(member_id),

    CONSTRAINT uk_follows UNIQUE (follower_id, following_id),

    CONSTRAINT ck_follows_self
        CHECK (follower_id <> following_id)
);
```

## 10. 알림 테이블 (notifications)

```sql
CREATE TABLE notifications (
    notification_id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    receiver_id     NUMBER NOT NULL,
    sender_id       NUMBER,
    post_id         NUMBER,
    comment_id      NUMBER,
    type            VARCHAR2(30) NOT NULL,
    message         VARCHAR2(500) NOT NULL,
    is_read         CHAR(1) DEFAULT 'N' NOT NULL,
    created_at      DATE DEFAULT SYSDATE NOT NULL,
    read_at         DATE,

    CONSTRAINT fk_notifications_receiver
        FOREIGN KEY (receiver_id) REFERENCES members(member_id),

    CONSTRAINT fk_notifications_sender
        FOREIGN KEY (sender_id) REFERENCES members(member_id),

    CONSTRAINT fk_notifications_post
        FOREIGN KEY (post_id) REFERENCES posts(post_id),

    CONSTRAINT fk_notifications_comment
        FOREIGN KEY (comment_id) REFERENCES comments(comment_id),

    CONSTRAINT ck_notifications_type
        CHECK (type IN ('COMMENT', 'LIKE', 'FOLLOW')),

    CONSTRAINT ck_notifications_read
        CHECK (is_read IN ('Y', 'N'))
);
```
