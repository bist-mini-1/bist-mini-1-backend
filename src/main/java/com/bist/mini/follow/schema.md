# 팔로우 테이블 스키마 (Follow Schema)

```sql
CREATE TABLE follows (
    follow_id       NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    follower_id     NUMBER NOT NULL,
    following_id    NUMBER NOT NULL,
    created_at      DATE DEFAULT SYSDATE NOT NULL,

    CONSTRAINT fk_follows_follower
        FOREIGN KEY (follower_id) REFERENCES members(member_id),

    CONSTRAINT fk_follows_following
        FOREIGN KEY (following_id) REFERENCES members(member_id),

    CONSTRAINT uk_follows UNIQUE (follower_id, following_id),

    CONSTRAINT ck_follows_self
        CHECK (follower_id <> following_id)
);
```
